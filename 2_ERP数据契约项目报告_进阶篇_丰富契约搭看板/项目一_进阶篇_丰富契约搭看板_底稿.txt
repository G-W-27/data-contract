### 💻 第三步：编写看板代码 `app.py`
在项目根目录下，新建一个文件，命名为 `app.py`。把下面这段完整的代码复制进去并保存：

```python
import streamlit as st
import pandas as pd
import json
import os

# 设置页面配置：宽屏模式，标题
st.set_page_config(page_title="ERP 数据质量监控看板", layout="wide")

st.title("📊 ERP 财务数据质量监控看板")
st.markdown("基于 Data Contract（数据契约）的自动化数据质量与内控审计报告")

# 检查并读取 JSON 报告
report_file = 'datacontract_report.json'
if not os.path.exists(report_file):
    st.error(f"❌ 找不到报告文件: {report_file}，请先运行 datacontract test 命令生成 JSON 报告。")
    st.stop()

with open(report_file, 'r', encoding='utf-8') as f:
    data = json.load(f)

# 提取检查结果列表 (根据 datacontract 的 JSON 结构适配)
checks = data.get('checks', [])
if not checks:
    st.warning("⚠️ 报告中没有找到检查记录，请检查 JSON 文件结构。")
    st.stop()

df = pd.DataFrame(checks)

# 兼容不同的 JSON 字段名 (由于版本差异，可能是 'result' 或 'status')
result_col = 'result' if 'result' in df.columns else 'status'
check_col = 'check' if 'check' in df.columns else 'name'

# 统计通过和失败数量
total_checks = len(df)
passed_checks = len(df[df[result_col] == 'passed'])
failed_checks = len(df[df[result_col] == 'failed'])

# 顶部卡片指标展示
col1, col2, col3 = st.columns(3)
col1.metric("📋 总检查项", total_checks)
col2.metric("✅ 通过项", passed_checks)
col3.metric("❌ 失败项", failed_checks, delta_color="inverse")

st.markdown("---")

# 饼图展示质量占比 (使用 Streamlit 自带图表)
st.subheader("📈 数据质量整体概况")
chart_data = pd.DataFrame({
    "状态": ["通过", "失败"],
    "数量": [passed_checks, failed_checks]
})
st.bar_chart(chart_data.set_index("状态"))

st.markdown("---")

# 详细数据表格展示
st.subheader("📋 详细检查结果")

# 定义高亮颜色函数
def highlight_result(val):
    if val == 'passed':
        return 'background-color: #d4edda; color: #155724;' # 绿色背景
    elif val == 'failed':
        return 'background-color: #f8d7da; color: #721c24;' # 红色背景
    return ''

# 展示带格式的表格
styled_df = df[[check_col, result_col]].style.map(highlight_result, subset=[result_col])
st.dataframe(styled_df, use_container_width=True)

st.success("🎉 数据契约校验完成！当前数据质量符合财务内控合规要求。")
```

### 📄 更新 `app.py` 适配新的 JSON 结构

`ci` 命令输出的 JSON 结构与 `test` 命令不同，字段名可能有变化。请用以下代码**完全替换**你 `app.py` 中的内容：

```python
import streamlit as st
import pandas as pd
import json
import os

# 页面配置
st.set_page_config(page_title="ERP 数据质量监控看板", layout="wide")

st.title("📊 ERP 财务数据质量监控看板")
st.markdown("基于 Data Contract（数据契约）的自动化数据质量与内控审计报告")

# 读取 JSON 报告
report_file = 'datacontract_report.json'
if not os.path.exists(report_file):
    st.error(f"❌ 找不到报告文件: {report_file}，请先运行 datacontract ci 命令生成 JSON 报告。")
    st.stop()

with open(report_file, 'r', encoding='utf-8') as f:
    data = json.load(f)

# --- 兼容不同版本的 JSON 结构 ---
# datacontract ci 的输出通常包含 "checks" 或 "results" 字段
checks = data.get('checks') or data.get('results') or []

if not checks:
    st.warning("⚠️ 报告中没有找到检查记录，请检查 JSON 文件结构。")
    st.json(data)  # 打印原始 JSON 方便调试
    st.stop()

df = pd.DataFrame(checks)

# 自动识别结果列和检查名称列
result_col = None
check_col = None
for col in ['result', 'status', 'state']:
    if col in df.columns:
        result_col = col
        break
for col in ['check', 'name', 'check_name', 'description']:
    if col in df.columns:
        check_col = col
        break

if not result_col or not check_col:
    st.error(f"❌ 无法识别 JSON 中的结果列，可用列: {list(df.columns)}")
    st.json(df.head())
    st.stop()

# 统计
total_checks = len(df)
passed_checks = len(df[df[result_col].astype(str).str.lower() == 'passed'])
failed_checks = len(df[df[result_col].astype(str).str.lower() == 'failed'])

# 顶部指标卡片
col1, col2, col3 = st.columns(3)
col1.metric("📋 总检查项", total_checks)
col2.metric("✅ 通过项", passed_checks)
col3.metric("❌ 失败项", failed_checks, delta_color="inverse")

st.markdown("---")

# 质量概况柱状图
st.subheader("📈 数据质量整体概况")
chart_data = pd.DataFrame({
    "状态": ["通过", "失败"],
    "数量": [passed_checks, failed_checks]
})
st.bar_chart(chart_data.set_index("状态"))

st.markdown("---")

# 详细结果表格
st.subheader("📋 详细检查结果")

def highlight_result(val):
    v = str(val).lower()
    if v == 'passed':
        return 'background-color: #d4edda; color: #155724;'
    elif v in ('failed', 'error'):
        return 'background-color: #f8d7da; color: #721c24;'
    return ''

display_df = df[[check_col, result_col]].copy()
display_df.columns = ['检查项', '结果']
styled_df = display_df.style.map(highlight_result, subset=['结果'])
st.dataframe(styled_df, use_container_width=True)

if failed_checks == 0:
    st.success("🎉 数据契约校验完成！当前数据质量符合财务内控合规要求。")
else:
    st.warning(f"⚠️ 发现 {failed_checks} 项检查未通过，请查看上方红色高亮行。")
```
